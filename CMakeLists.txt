cmake_minimum_required(VERSION 3.20)
project(mini_bio_di LANGUAGES CXX)

add_library(mini INTERFACE)
target_include_directories(mini INTERFACE ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_compile_features(mini INTERFACE cxx_std_23)

option(MINI_SANITIZE "Enable AddressSanitizer and UndefinedBehaviorSanitizer" OFF)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(mini INTERFACE -Wall -Wextra -Wpedantic -Werror)
    if(MINI_SANITIZE)
        target_compile_options(mini INTERFACE -fsanitize=address,undefined -fno-omit-frame-pointer)
        target_link_options(mini INTERFACE -fsanitize=address,undefined)
    endif()
endif()

add_executable(example examples/main.cpp)
target_link_libraries(example PRIVATE mini)
add_executable(implicit_example examples/implicit_scope.cpp)
target_link_libraries(implicit_example PRIVATE mini)

enable_testing()
foreach(name IN ITEMS bio_test io_test di_test stack_safety example_test syntax_test accessor_test implicit_test do_test do_stack_safety)
    add_executable(${name} tests/${name}.cpp)
    target_link_libraries(${name} PRIVATE mini)
    add_test(NAME ${name} COMMAND ${name})
endforeach()
add_test(NAME example_live COMMAND example live)
set_tests_properties(example_live PROPERTIES PASS_REGULAR_EXPRESSION "Hello, Ada!")
add_test(NAME example_cli_test COMMAND example test)
set_tests_properties(example_cli_test PROPERTIES PASS_REGULAR_EXPRESSION "Welcome, Test Ada.")
add_test(NAME implicit_example COMMAND implicit_example)
set_tests_properties(implicit_example PROPERTIES PASS_REGULAR_EXPRESSION "IO: 42 [(]3 local pure calls[)]")
